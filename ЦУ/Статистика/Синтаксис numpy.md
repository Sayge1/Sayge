```python
np.array.ndim # сколько измерений в массиве
np.array.shape # `[[1,2,3],[1,2,3]]` команда выдаст (2, 3) (количество элементов в каждом пространстве)
np.array.reshape # меняет размерность массива (например с (1) на (2, 3))
.reshape(-1, 2) # автоматически подберет подходящий параметр для решейпа вместо -1

np.array.astype(int) # превращает элементы массива в инты (они изначально флоаты)
np.array.dtype # вывести тип элементов в массиве

np.full((5,5), 23)  # заполнение матрицы 5 на 5 с помощью 23


np.mean(np.array) - среднее арифметическое
np.var(np.array) - дисперсия
```
