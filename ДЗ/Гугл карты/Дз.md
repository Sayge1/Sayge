INF = float("inf")

def djikstra(graph, start, finish):
    n_vertices = len(graph)
    dists = [INF] * n_vertices
    visited = [False] * n_vertices
    dists[start] = 0
    expanded = 0
    while True:
        min_v = -1
        for v in range(n_vertices):
            if visited[v]:
                continue
            if min_v == -1 or dists[v] < dists[min_v]:
                min_v = v
        if min_v == -1 or dists[min_v] == INF:
            break

        visited[min_v] = True
        expanded += 1

        if min_v == finish:
            break

        for edge in graph[min_v]:
            if dists[edge.to] > dists[min_v] + edge.weight:
                dists[edge.to] = dists[min_v] + edge.weight
    return dists, expanded

dists, expanded = djikstra(graph, start, finish)
